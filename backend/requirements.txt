# ============================================================
# 后端依赖 —— AI 智能知识库 Agent 平台
# 安装：cd backend && pip install -r requirements.txt
# 当前阶段包含骨架 + PostgreSQL 接入 + 数据库迁移 + DeepSeek LLM 接入所需依赖，
# 向量库（Milvus）待下一步向量化功能开发时再加入。
# ============================================================

# Web 框架：提供路由、依赖注入与自动生成 OpenAPI 文档
fastapi>=0.141.1

# multipart/form-data 解析器：FastAPI 处理文件上传（UploadFile / Form）的必需依赖。
# 它不在 FastAPI 的默认依赖里，漏装时不是「上传功能不可用」这么轻 ——
# FastAPI 在「定义路由」的时候就会检查它，缺了会直接抛 RuntimeError，
# 导致整个应用导入失败。所以只要有上传接口，这个包就必须在。
python-multipart>=0.0.32

# ASGI 服务器：实际跑起 FastAPI 应用的进程
# [standard] 附带 httptools / uvloop / watchfiles，--reload 热重载依赖它
uvicorn[standard]>=0.30.0

# 配置管理：从 .env 与环境变量读取配置并做类型校验
# 选它而不是自己读 os.environ，是为了让配置有类型、有默认值、缺失时能给出明确报错
pydantic-settings>=2.14.2

# ---------- 数据库 ----------

# ORM：用 SQLAlchemy 2.x 的类型化声明式写法（Mapped / mapped_column），
# 让「Python 类型提示」和「数据库列定义」写在一起。
# [asyncio] 这个 extra 会自动装上 greenlet —— SQLAlchemy 异步支持的底层依赖，
# 单独装 sqlalchemy 而漏掉它，运行时会报 MissingGreenlet。
sqlalchemy[asyncio]>=2.0.54

# PostgreSQL 异步驱动：SQLAlchemy 通过 postgresql+asyncpg 这个方言调用它。
# 选 asyncpg 而不是同步的 psycopg2，是因为 FastAPI 跑在事件循环上，
# 同步驱动执行查询时会阻塞整个事件循环，导致其他请求排队等待。
asyncpg>=0.31.0

# 数据库迁移工具：把「模型定义的改动」变成一份份可版本化、可回放的脚本。
# 选它而不是自己写 CREATE TABLE / 建表 SQL：迁移脚本进版本库后，
# 任何人 clone 下来执行 alembic upgrade head 就能得到完全一致的库结构，
# 改了字段也能回滚，不用手工比对每个人的库差在哪。
# 它跑在 SQLAlchemy 之上（已在上面安装），另外会自动带上 Mako 模板依赖。
alembic>=1.13

# ---------- 大模型 ----------

# DeepSeek 官方提供的是「OpenAI 兼容」接口（请求体、响应结构和 OpenAI 一致），
# 所以直接复用 openai 官方 SDK，只把 base_url 指向 https://api.deepseek.com 即可，
# 没必要再引入一个 DeepSeek 专用 SDK。
#
# 注意这里用的是 AsyncOpenAI 而不是同步的 OpenAI：
# FastAPI 跑在事件循环上，一次模型调用要等好几秒，同步 SDK 会把这期间
# 整个事件循环堵死，导致其他请求只能排队；async 版本则是把等待时间让出去。
openai>=3.11.0

# ---------- 向量数据库 ----------

# Milvus 官方 Python 客户端。选它而不是自己拼 gRPC / HTTP 请求：
# 建集合、定义索引、插入、向量检索这些操作，参数都要按 Milvus 的 proto 结构组织，
# 官方 SDK 已经把它们封装成 Python 方法，并代管连接、重试和类型转换。
#
# 版本必须和服务端对齐主版本：本地容器跑的是 milvusdb/milvus:v3.0.1，
# 所以这里用 3.0.x 的客户端。用 2.x 客户端去连 3.0 服务端会因为接口不兼容而连不上。
pymilvus>=3.0.2
